rules_version = '2';
service cloud.firestore {
  match /databases/{d}/documents {
    function admin() { return request.auth != null && (request.auth.token.phone_number == '+9647504211356' || request.auth.token.email == 'zeravan180@gmail.com'); }
    match /ads/{id} {
      allow read: if true;
      allow create: if request.auth != null && request.resource.data.uid == request.auth.uid && request.resource.data.vip == false;
      allow update: if admin()
        || (request.auth != null && resource.data.uid == request.auth.uid && request.resource.data.uid == resource.data.uid && request.resource.data.vip == resource.data.vip)
        || (request.auth != null && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['views']) && request.resource.data.views == resource.data.get('views', 0) + 1);
      allow delete: if admin() || (request.auth != null && resource.data.uid == request.auth.uid);
    }
    match /cvs/{u} { allow read, write: if request.auth != null && request.auth.uid == u; }
    match /rates/{id} { allow read: if true; allow write: if admin(); }
  }
}
