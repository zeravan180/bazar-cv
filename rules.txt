    function admin() { return request.auth != null && (request.auth.token.phone_number == '+9647504211356' || request.auth.token.email == 'zeravan180@gmail.com'); }
    match /ads/{id} {
      allow read: if true;
      allow create: if request.auth != null && request.resource.data.uid == request.auth.uid && request.resource.data.vip == false && !('vipUntil' in request.resource.data) && !exists(/databases/$(d)/documents/banned/$(request.auth.uid));
